# Rosalind: Mendel's First Law
# Probability that two randomly selected mating organisms produce
# offspring with a dominant phenotype
k = 2  # homozygous dominant
m = 2  # heterozygous
n = 2  # homozygous recessive

total = k + m +n
# Probability of picking two of each combination (without replacement)
prob_aa_aa = (n / total) * ((n - 1) / total - 1))
prob_aa_Aa = (n / total) * m / (total - 1) + (m / total) * n / (total - 1)
prob_Aa_Aa = (m / total) * ((m - 1) / total - 1)

# Recessive offspring probability from each pairing
prob_recessive = prob_aa_aa * 1 + prob_aa_Aa * 0.5 + prob_Aa_Aa
prob_dominant = 1 - prob_recessive
print(round(prob_dominant, 5)
